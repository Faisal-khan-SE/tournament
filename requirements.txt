# ---------------------------------------------------------------------------
# Environment variables (see .env.example)
# ---------------------------------------------------------------------------
# ADMIN_PASSWORD   organiser/scorer password. Unset = site runs open (private
#                  network only). Set = every write needs a sign-in at /login,
#                  or an "Authorization: Bearer <password>" header.
# NODE_ENV         "production" for `npm start`; Prisma query logging is on in
#                  development only.
# PORT             optional; `next start -p <port>` also works.
#
# The database is a single SQLite file at prisma/dev.db (path fixed in
# prisma/schema.prisma); no DATABASE_URL is needed.

# ---------------------------------------------------------------------------
# Setup
# ---------------------------------------------------------------------------
# npm install
# cp .env.example .env         # then set ADMIN_PASSWORD
# npm run setup                # prisma generate + db push + seed (wipes the DB)
# npm run dev                  # http://localhost:3000
#
# Production:
# npm run build && npm start
#
# Verify:
# npm run typecheck            # tsc --noEmit
# npm test                     # integration suites against a running server
#                              # (ADMIN_PASSWORD=<pw> npm test when auth is on)
